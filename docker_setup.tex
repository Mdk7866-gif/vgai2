\section{Files to Add in Your Source Code}

These files go \textbf{inside your Git repository} and are used to build Docker images.

% ── backend/Dockerfile ───────────────────────────────────────────────────────
\subsection{backend/Dockerfile}

\begin{codeblock}
\begin{lstlisting}[style=codestyle]
# Use Python 3.13 slim image
FROM python:3.13-slim

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1
ENV PYTHONPATH=/app

WORKDIR /app

# Copy metadata files first (layer caching)
COPY pyproject.toml README.md ./

# Install Python dependencies
RUN pip install --no-cache-dir hatchling && \
    pip install --no-cache-dir .

# Copy the rest of the application
COPY . .

EXPOSE 8000

CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000"]
\end{lstlisting}
\end{codeblock}

% ── frontendweb/Dockerfile ───────────────────────────────────────────────────
\subsection{frontendweb/Dockerfile}

\begin{codeblock}
\begin{lstlisting}[style=codestyle]
# Stage 1: Build
FROM node:20-alpine AS builder
WORKDIR /app
COPY package.json package-lock.json* ./
RUN npm install
COPY . .
RUN npm run build

# Stage 2: Run
FROM node:20-alpine AS runner
WORKDIR /app
ENV NODE_ENV=production
COPY --from=builder /app/package.json ./
COPY --from=builder /app/package-lock.json* ./
COPY --from=builder /app/next.config.ts ./
COPY --from=builder /app/public ./public
COPY --from=builder /app/.next ./.next
COPY --from=builder /app/node_modules ./node_modules
EXPOSE 3000
CMD ["npm", "run", "start"]
\end{lstlisting}
\end{codeblock}

\section{Build \& Push Images to Docker Hub (You Do This)}

Run from the \textbf{root of your project}. Do this every time you update your code.

\subsection{Step 1 — Login}
\begin{codeblock}
\begin{lstlisting}[style=commandstyle]
docker login
# Enter username: mdk7866  and your password
\end{lstlisting}
\end{codeblock}

\subsection{Step 2 — Build and Push Backend}
\begin{codeblock}
\begin{lstlisting}[style=commandstyle]
docker build -t mdk7866/vgai-backend:latest ./backend
docker push  mdk7866/vgai-backend:latest
\end{lstlisting}
\end{codeblock}

\subsection{Step 3 — Build and Push Frontend}
\begin{warningbox}
\textbf{Important:} Next.js bakes \texttt{NEXT\_PUBLIC\_} variables into the static build at \textbf{build time}. Make sure \texttt{NEXT\_PUBLIC\_BACKEND\_URL=http://localhost:8000} is set in \texttt{frontendweb/.env} before building.
\end{warningbox}
\begin{codeblock}
\begin{lstlisting}[style=commandstyle]
docker build -t mdk7866/vgai-frontend:latest ./frontendweb
docker push  mdk7866/vgai-frontend:latest
\end{lstlisting}
\end{codeblock}

Create a new empty folder, put \textbf{exactly these 3 files} inside, and send/zip it:

\begin{center}
\noindent\begin{minipage}{0.55\linewidth}
\begin{mdframed}[linecolor=sectioncolor,linewidth=1.5pt,backgroundcolor=stepbg,

\subsection{docker-compose.yml (goes in sharing folder)}
\begin{codeblock}
\begin{lstlisting}[style=codestyle]
services:
  backend:
    image: mdk7866/vgai-backend:latest
    container_name: vgai-backend
    ports:
      - "8000:8000"
    volumes:
      - ./downloads:/app/downloads
    env_file:
      - .env.backend
    restart: always

  frontend:
    image: mdk7866/vgai-frontend:latest
    container_name: vgai-frontend
    ports:
      - "3000:3000"
    env_file:
      - .env.frontend
    depends_on:
      - backend
    restart: always

networks:
  default:
    name: vgai-network
\end{lstlisting}
\end{codeblock}

\subsection{.env.backend (copy from backend/.env)}
\begin{codeblock}
\begin{lstlisting}[style=codestyle]

CLOUDINARY_API_KEY=your_key
CLOUDINARY_API_SECRET=your_secret
GEMINI_FREE_API_KEY_MUJAMAHE=your_gemini_key
\end{lstlisting}
\end{codeblock}

\subsection{.env.frontend (copy from frontendweb/.env)}
\begin{codeblock}
\begin{lstlisting}[style=codestyle]
NEXT_PUBLIC_BACKEND_URL=http://localhost:8000
\end{lstlisting}
\end{codeblock}

\subsection{Instructions for Your Friend}
\begin{stepbox}
\begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}, itemsep=3pt]
  \item Install \textbf{Docker Desktop} from \texttt{https://www.docker.com/products/docker-desktop}
  \item Unzip the folder you received
  \item Open a terminal \textbf{inside} that folder
  \item Run: \quad\texttt{\color{sectioncolor}docker-compose up -d}
  \item Open browser $\rightarrow$ \texttt{http://localhost:3000} \checkmark
  \item To stop: \quad\texttt{\color{sectioncolor}docker-compose down}
\end{enumerate}
\end{stepbox}

\section{Docker Commands Cheat Sheet}

\subsection{Start / Stop}
\begin{codeblock}
\begin{lstlisting}[style=commandstyle]
docker-compose up -d            # start all (background)
docker-compose down             # stop and remove containers
docker-compose restart vgai-backend  # restart one service
\end{lstlisting}
\end{codeblock}

\subsection{Build \& Push}
\begin{codeblock}
\begin{lstlisting}[style=commandstyle]
docker build -t mdk7866/vgai-backend:latest  ./backend
docker build -t mdk7866/vgai-frontend:latest ./frontendweb
docker push mdk7866/vgai-backend:latest
docker push mdk7866/vgai-frontend:latest
\end{lstlisting}
\end{codeblock}

\subsection{Logs \& Status}
\begin{codeblock}
\begin{lstlisting}[style=commandstyle]
docker ps                        # list running containers
docker logs -f vgai-backend      # live backend logs
docker logs -f vgai-frontend     # live frontend logs
docker system df                 # disk usage
\end{lstlisting}
\end{codeblock}

\subsection{Cleanup}
\begin{codeblock}
\begin{lstlisting}[style=commandstyle]
docker system prune -a           # remove all unused images & cache
docker rmi mdk7866/vgai-backend:latest  # remove a specific image
docker pull mdk7866/vgai-backend:latest # pull latest from Docker Hub
\end{lstlisting}
\end{codeblock}

\begin{tipbox}
\textbf{Tip:} Run \texttt{docker system prune -a} occasionally to free up disk space from old cached build layers.
\end{tipbox}
